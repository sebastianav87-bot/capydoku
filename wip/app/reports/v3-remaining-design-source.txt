window.CAPY = {
  zones: [
    ["A","A","A","B","B","C"],
    ["A","B","B","B","B","C"],
    ["A","A","B","E","E","C"],
    ["D","D","D","E","C","C"],
    ["D","E","E","E","C","F"],
    ["D","D","E","F","F","F"]
  ],
  rocks: new Set(["0,2"]),
  solution: new Set(["0,1","1,3","2,5","3,0","4,2","5,4"]),
  placed: new Set(["0,1","3,0"]),
  xs: new Set(["0,0","0,4","1,1","2,1","4,4","5,1"]),
  colors: {
    A:["#6CC874","#5EBA66"],
    B:["#BBE2F9","#ADD4EB"],
    C:["#F9DF88","#EBD17A"],
    D:["#AE7ED0","#A070C2"],
    E:["#F1977B","#E3896D"],
    F:["#5FC6BF","#51B8B1"]
  },
  cues: {A:'dots',B:'hdashes',C:'lily',D:'diag',E:'rings',F:'vdashes'}
};

function patternSVG(kind, stroke){
  if(kind==='dots') return `<svg class="pat" viewBox="0 0 60 60" aria-hidden="true"><circle cx="15" cy="15" r="2.2" fill="${stroke}"/><circle cx="45" cy="18" r="2.2" fill="${stroke}"/><circle cx="28" cy="42" r="2.2" fill="${stroke}"/></svg>`;
  if(kind==='hdashes') return `<svg class="pat" viewBox="0 0 60 60"><path d="M10 18h16M34 18h16M12 40h14M36 40h14" stroke="${stroke}" stroke-width="2.2" stroke-linecap="round"/></svg>`;
  if(kind==='lily') return `<svg class="pat" viewBox="0 0 60 60"><ellipse cx="18" cy="22" rx="7" ry="3.5" fill="${stroke}" transform="rotate(-20 18 22)"/><ellipse cx="42" cy="40" rx="6" ry="3" fill="${stroke}" transform="rotate(25 42 40)"/></svg>`;
  if(kind==='diag') return `<svg class="pat" viewBox="0 0 60 60"><path d="M14 46l12-12M30 48l14-14M12 28l10-10" stroke="${stroke}" stroke-width="2.2" stroke-linecap="round"/></svg>`;
  if(kind==='rings') return `<svg class="pat" viewBox="0 0 60 60"><circle cx="20" cy="22" r="6" fill="none" stroke="${stroke}" stroke-width="2"/><circle cx="42" cy="40" r="5" fill="none" stroke="${stroke}" stroke-width="2"/></svg>`;
  if(kind==='vdashes') return `<svg class="pat" viewBox="0 0 60 60"><path d="M18 10v16M18 34v16M42 12v14M42 36v14" stroke="${stroke}" stroke-width="2.2" stroke-linecap="round"/></svg>`;
  return '';
}

function buildBoard(opts){
  const {mode, gridEl} = opts;
  const Z = CAPY.zones;
  gridEl.innerHTML = '';
  gridEl.classList.add('zone-grid');

  for(let r=0;r<6;r++){
    for(let c=0;c<6;c++){
      const z=Z[r][c];
      const cell=document.createElement('div');
      const [hi,lo]=CAPY.colors[z];
      cell.className='cell';
      cell.style.background=`linear-gradient(145deg,${hi},${lo})`;
      const key=r+','+c;
      cell.innerHTML = patternSVG(CAPY.cues[z], 'rgba(255,255,255,0.40)');
      if(CAPY.rocks.has(key)){
        cell.insertAdjacentHTML('beforeend','<img class="rock" src="../art/rock.png" alt="rock"/>');
      } else if(mode==='solved' && CAPY.solution.has(key)){
        cell.insertAdjacentHTML('beforeend','<img class="capy" src="../art/capy-jump.png"/>');
      } else if(mode==='progress' && CAPY.placed.has(key)){
        cell.insertAdjacentHTML('beforeend','<img class="capy" src="../art/capy-sit.png"/>');
      } else if(mode==='progress' && CAPY.xs.has(key)){
        cell.insertAdjacentHTML('beforeend','<div class="xmark">×</div>');
      }
      gridEl.appendChild(cell);
    }
  }
  requestAnimationFrame(()=>drawOutlines(gridEl));
}

window.drawOutlines = function drawOutlines(gridEl){
  gridEl.querySelectorAll('svg.zone-outline').forEach(e=>e.remove());
  const cells=[...gridEl.querySelectorAll('.cell')];
  if(cells.length!==36) return;
  const gr=gridEl.getBoundingClientRect();
  const c00=cells[0].getBoundingClientRect();
  const c01=cells[1].getBoundingClientRect();
  const c10=cells[6].getBoundingClientRect();
  const cellW=c00.width, cellH=c00.height;
  const gapX=c01.left-c00.right, gapY=c10.top-c00.bottom;
  const ox=c00.left-gr.left, oy=c00.top-gr.top;
  const W=gr.width, H=gr.height;
  const SW=5;
  const half=SW/2;
  const xs=[], ys=[];
  xs[0] = ox - gapX/2;
  ys[0] = oy - gapY/2;
  for(let i=1;i<=5;i++){
    xs[i] = ox + i*(cellW+gapX) - gapX/2;
    ys[i] = oy + i*(cellH+gapY) - gapY/2;
  }
  xs[6] = ox + 6*cellW + 5*gapX + gapX/2;
  ys[6] = oy + 6*cellH + 5*gapY + gapY/2;
  for(let i=0;i<=6;i++){
    xs[i] = Math.min(W-half, Math.max(half, xs[i]));
    ys[i] = Math.min(H-half, Math.max(half, ys[i]));
  }
  const svgNS='http://www.w3.org/2000/svg';
  const svg=document.createElementNS(svgNS,'svg');
  svg.setAttribute('class','zone-outline');
  svg.setAttribute('width',String(W));
  svg.setAttribute('height',String(H));
  svg.setAttribute('viewBox',`0 0 ${W} ${H}`);
  svg.style.cssText='position:absolute;left:0;top:0;width:100%;height:100%;pointer-events:none;z-index:8;overflow:hidden;';
  const g=document.createElementNS(svgNS,'g');
  g.setAttribute('fill','none');
  g.setAttribute('stroke','#45554F');
  g.setAttribute('stroke-width',String(SW));
  g.setAttribute('stroke-linejoin','round');
  g.setAttribute('stroke-linecap','butt');
  const outer=document.createElementNS(svgNS,'rect');
  outer.setAttribute('x',String(xs[0]));
  outer.setAttribute('y',String(ys[0]));
  outer.setAttribute('width',String(xs[6]-xs[0]));
  outer.setAttribute('height',String(ys[6]-ys[0]));
  outer.setAttribute('rx','11');
  outer.setAttribute('ry','11');
  g.appendChild(outer);
  const Z=CAPY.zones;
  for(let r=0;r<6;r++){
    for(let c=0;c<5;c++){
      if(Z[r][c]===Z[r][c+1]) continue;
      let y1=ys[r], y2=ys[r+1];
      if(r===0) y1 = ys[0] + half;
      if(r===5) y2 = ys[6] - half;
      const line=document.createElementNS(svgNS,'line');
      line.setAttribute('x1',String(xs[c+1]));
      line.setAttribute('x2',String(xs[c+1]));
      line.setAttribute('y1',String(y1));
      line.setAttribute('y2',String(y2));
      g.appendChild(line);
    }
  }
  for(let r=0;r<5;r++){
    for(let c=0;c<6;c++){
      if(Z[r][c]===Z[r+1][c]) continue;
      let x1=xs[c], x2=xs[c+1];
      if(c===0) x1 = xs[0] + half;
      if(c===5) x2 = xs[6] - half;
      const line=document.createElementNS(svgNS,'line');
      line.setAttribute('x1',String(x1));
      line.setAttribute('x2',String(x2));
      line.setAttribute('y1',String(ys[r+1]));
      line.setAttribute('y2',String(ys[r+1]));
      g.appendChild(line);
    }
  }
  svg.appendChild(g);
  gridEl.appendChild(svg);
};

#!/usr/bin/env python3
"""Capydoku 6x6 uniqueness verifier. Enumerates ALL solutions under the rules."""
from itertools import permutations
from collections import defaultdict

N = 6

# Zone labels A-F (unequal sizes OK; exactly one capy per zone)
ZONES = [
    list("AAABBC"),
    list("ABBBBC"),
    list("AABEEC"),
    list("DDDECC"),
    list("DEEECF"),
    list("DDEFFF"),
]

ROCKS = {(0, 2)}

# Known unique solution (row, col), 0-indexed
SOLUTION = [(0, 1), (1, 3), (2, 5), (3, 0), (4, 2), (5, 4)]

# Mockup progress state
PLACED = [(0, 1), (3, 0)]  # must be in solution
X_MARKS = [(0, 0), (0, 4), (1, 1), (2, 1), (4, 4), (5, 1)]  # must NOT be in solution


def king_adjacent(a, b):
    r1, c1 = a
    r2, c2 = b
    return max(abs(r1 - r2), abs(c1 - c2)) <= 1 and a != b


def valid(perm):
    cells = [(r, perm[r]) for r in range(N)]
    if any(cell in ROCKS for cell in cells):
        return False
    zc = defaultdict(int)
    for r, c in cells:
        zc[ZONES[r][c]] += 1
    if len(zc) != N or any(v != 1 for v in zc.values()):
        return False
    for i in range(N):
        for j in range(i + 1, N):
            if king_adjacent(cells[i], cells[j]):
                return False
    return True


def enumerate_all():
    return [[(r, p[r]) for r in range(N)] for p in permutations(range(N)) if valid(p)]


def main():
    sols = enumerate_all()
    lines = []
    lines.append("Capydoku 6x6 uniqueness verifier")
    lines.append("Rules: one per row, column, zone; no orthogonal/diagonal touch; rocks blocked")
    lines.append("ZONES:")
    for row in ZONES:
        lines.append("  " + " ".join(row))
    lines.append(f"ROCKS: {sorted(ROCKS)}")
    lines.append(f"solutions_found: {len(sols)}")
    for i, s in enumerate(sols):
        lines.append(f"solution[{i}]: {s}")
    unique = len(sols) == 1
    lines.append(f"unique: {unique}")
    ok = True
    if unique:
        sol = set(map(tuple, sols[0]))
        expected = set(SOLUTION)
